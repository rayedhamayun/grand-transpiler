public class Prog {
    public static void run() {
        for (int i = 0; i < 12; i++) {
            if (i == 2) {
                continue;
            }
            if (i % 2 == 0 && i > 6) {
                System.out.print("even ");
            } else if (i % 3 == 0 || i == 1) {
                System.out.print("three ");
            } else {
                System.out.print("other ");
            }
            if (i == 10) {
                break;
            }
        }
        System.out.println();
        int n = 3;
        while (n > 0) {
            System.out.println(n);
            n--;
        }
        do {
            System.out.println(n);
            n++;
        } while (n < 2);
        for (int i = 0, j = 10; i < j; i += 3, j -= 2) {
            System.out.println(i * j);
        }
        int k = 0;
        for (; ;) {
            k++;
            if (k > 3) {
                break;
            }
        }
        System.out.println(k > 3 ? 1 : 0);
        final boolean flag = !(k == 4);
        if (!flag) {
            System.out.println("flag off");
        }
        while (true) {
            for (int i = 0; i < 3; i++) {
                if (i == 1) {
                    break;
                }
                System.out.println(i);
            }
            break;
        }
    }
}
