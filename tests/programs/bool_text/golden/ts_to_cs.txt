using System;

public class Prog
{
    public static void run()
    {
        bool yes = 1 < 2;
        Console.WriteLine(yes ? "true" : "false");
        Console.WriteLine("value: " + (yes ? "true" : "false"));
        Console.WriteLine("interpolated " + (!yes ? "true" : "false") + " " + (1 + 1) + 2);
    }
}
